196std::vector<char> VulkanPipeline::readFile(const std::filesystem::path& filepath)
197{
198 std::ifstream file(filepath, std::ios::ate | std::ios::binary);
199
200 if (!file.is_open())
201 throwWithLog<FileIOException>(std::source_location::current(), filepath);
202
203 const std::streamsize fileSize{ file.tellg() };
204 std::vector<char> buffer(fileSize);
205
206 file.seekg(0);
207 file.read(buffer.data(), fileSize);
208
209 file.close();
210
211 return buffer;
212}
