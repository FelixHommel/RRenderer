59void VulkanRenderer::render()
60{
61 // TODO: find better way to limit frames
62 constexpr int FPS{ 33 };
63 std::this_thread::sleep_for(std::chrono::milliseconds(FPS));
64
65 std::uint32_t imageIndex{};
66 auto result{ m_swapchain->acquireNextImage(&imageIndex) };
67
68 if (result == VK_ERROR_OUT_OF_DATE_KHR)
69 {
70 recreateSwapchain();
71
72 return;
73 }
74
75 if (result != VK_SUCCESS && result != VK_SUBOPTIMAL_KHR)
76 throwWithLog<VulkanException>(
77 std::source_location::current(), VulkanExceptionCause::IMAGE_ACQUISITION);
78
79 recordCommandBuffers(imageIndex);
80 result = m_swapchain->submitCommandBuffer(&m_commandBuffers[imageIndex]->getHandleRef(), &imageIndex);
81
82 if (result == VK_ERROR_OUT_OF_DATE_KHR || result == VK_SUBOPTIMAL_KHR || window.wasWindowResized())
83 {
84 window.resetWindowResized();
85 recreateSwapchain();
86
87 return;
88 }
89
90 if (result != VK_SUCCESS)
91 throwWithLog<VulkanException>(
92 std::source_location::current(), VulkanExceptionCause::SUBMIT_COMMAND_BUFFER);
93}

103void VulkanRenderer::loadRenderObjects()
104{
105 const std::vector<Vertex> vertices{
106 { .position = { 0.f, -0.5f }, .color = { 1.f, 0.f, 0.f } }, //NOLINT
107 { .position = { 0.5f, 0.5f }, .color = { 0.f, 1.f, 0.f } }, //NOLINT
108 { .position = { -0.5f, 0.5f }, .color = { 0.f, 0.f, 1.f } } //NOLINT
109 };
110 const auto mesh = std::make_shared<VulkanMesh>(*m_device, vertices);
111
112 std::vector<glm::vec3> colors{
113 { 1.f, 0.7f, 0.73f }, //NOLINT
114 { 1.f, 0.87f, 0.73f }, //NOLINT
115 { 1.f, 1.f, 0.73f }, //NOLINT
116 { 0.73f, 1.f, 0.8f }, //NOLINT
117 { 0.73f, 0.88f, 1.f } //NOLINT
118 };
119
120 for (auto& color : colors)
121 color = glm::pow(color, glm::vec3(2.2f)); //NOLINT
122
123 constexpr std::size_t triCount{ 40 };
124 for (std::size_t i{ 0 }; i < triCount; ++i)
125 {
126 RenderObject triangle{ mesh,
127 colors.at(i % colors.size()),
128 { .translation = {},
129 .scale = { glm::vec2(0.5f) + i * 0.025f },
130 .rotation = i * std::numbers::pi_v<float> * 0.025f } }; //NOLINT
131
132 m_renderObjects.push_back(std::move(triangle));
133 }
134}

157void VulkanRenderer::recreateSwapchain()
158{
159 auto extent{ window.getExtent() };
160 while (extent.height == 0 || extent.width == 0)
161 {
162 extent = window.getExtent();
163 glfwWaitEvents();
164 }
165
166 vkDeviceWaitIdle(m_device->getHandle());
167
168 if (m_swapchain == nullptr)
169 {
170 m_swapchain = std::make_unique<VulkanSwapchain>(*m_device, m_surface->getHandle(), extent);
171 }
172 else
173 {
174 m_swapchain = std::make_unique<VulkanSwapchain>(
175 *m_device, m_surface->getHandle(), extent, std::move(m_swapchain));
176
177 if (m_swapchain->imageCount() != m_commandBuffers.size())
178 m_commandBuffers = m_commandPool->allocateCommandBuffer(m_swapchain->imageCount());
179 }
180
181 m_pipeline = createPipeline();
182}

187void VulkanRenderer::recordCommandBuffers(std::size_t imageIndex)
188{
189 const VkCommandBufferBeginInfo beginInfo{ .sType = VK_STRUCTURE_TYPE_COMMAND_BUFFER_BEGIN_INFO };
190
191 if (vkBeginCommandBuffer(m_commandBuffers.at(imageIndex)->getHandleRef(), &beginInfo) != VK_SUCCESS)
192 throwWithLog<VulkanException>(
193 std::source_location::current(), VulkanExceptionCause::BEGIN_RECORD_COMMAND_BUFFER, imageIndex);
194
195 constexpr std::array<VkClearValue, 2> clearValues{ VkClearValue{ .color = CLEAR_COLOR },
196 VkClearValue{ .depthStencil = { 1.f, 0 } } };
197 const VkRenderPassBeginInfo renderPassBeginInfo{
198 .sType = VK_STRUCTURE_TYPE_RENDER_PASS_BEGIN_INFO,
199 .renderPass = m_swapchain->getRenderPassHandle(),
200 .framebuffer = m_swapchain->getFramebufferHandle(imageIndex),
201 .renderArea = { .offset = { 0, 0 }, .extent = m_swapchain->getExtent() },
202 .clearValueCount = static_cast<std::uint32_t>(clearValues.size()),
203 .pClearValues = clearValues.data()
204 };
205
206 vkCmdBeginRenderPass(
207 m_commandBuffers[imageIndex]->getHandleRef(), &renderPassBeginInfo, VK_SUBPASS_CONTENTS_INLINE);
208
209 const VkViewport viewport{ .x = 0,
210 .y = 0,
211 .width = static_cast<float>(m_swapchain->getExtent().width),
212 .height = static_cast<float>(m_swapchain->getExtent().height),
213 .minDepth = 0.f,
214 .maxDepth = 1.f };
215 vkCmdSetViewport(m_commandBuffers[imageIndex]->getHandleRef(), 0, 1, &viewport);
216
217 const VkRect2D scissor{ { 0, 0 }, m_swapchain->getExtent() };
218 vkCmdSetScissor(m_commandBuffers[imageIndex]->getHandleRef(), 0, 1, &scissor);
219
220 renderObjects(m_commandBuffers[imageIndex]->getHandleRef());
221
222 vkCmdEndRenderPass(m_commandBuffers[imageIndex]->getHandleRef());
223
224 if (vkEndCommandBuffer(m_commandBuffers[imageIndex]->getHandleRef()) != VK_SUCCESS)
225 throwWithLog<VulkanException>(
226 std::source_location::current(), VulkanExceptionCause::END_RECORD_COMMAND_BUFFER, imageIndex);
227}

232void VulkanRenderer::renderObjects(VkCommandBuffer commandBuffer)
233{
234 int i{ 0 };
235 for (auto& obj : m_renderObjects)
236 {
237 ++i;
238 obj.setRotation(
239 glm::mod<float>(obj.getRotation() + 0.001f * i, 2.f * std::numbers::pi_v<float>)); //NOLINT
240 }
241
242 m_pipeline->bind(commandBuffer);
243 for (auto& obj : m_renderObjects)
244 {
245 SimplePushConstantData pushData{ .transform = obj.getTransformMatrix(),
246 .offset = obj.getTranslation(),
247 .color = obj.getColor() };
248
249 vkCmdPushConstants(
250 commandBuffer,
251 m_pipelineLayout->getHandle(),
252 VK_SHADER_STAGE_VERTEX_BIT | VK_SHADER_STAGE_FRAGMENT_BIT,
253 0,
254 sizeof(SimplePushConstantData),
255 &pushData);
256
257 obj.render(commandBuffer);
258 }
259}
