131void VulkanDevice::pickPhysicalDevice()
132{
133 std::uint32_t deviceCount{0};
134 vkEnumeratePhysicalDevices(instance, &deviceCount, nullptr);
135
136 if(deviceCount == 0)
137 throwWithLog<VulkanException>(std::source_location::current(), VulkanExceptionCause::NO_PHYSICAL_DEVICE_FOUND);
138
139 spdlog::info("Found {} vulkan compatible device(s)", deviceCount);
140
141 std::vector<VkPhysicalDevice> devices(deviceCount);
142 vkEnumeratePhysicalDevices(instance, &deviceCount, devices.data());
143
144 for(const auto& dev : devices)

157 spdlog::info("physical device: {}\n\tID: {}\n\tvendorID: {}\n\tdeviceType: {}",

167void VulkanDevice::createLogicalDevice()
168{
169 QueueFamilyIndices indices{ findQueueFamilies(m_physicalDevice) };
170
171 std::vector<VkDeviceQueueCreateInfo> queueCreateInfos;
172 std::set uniqueQueueFamilies{ indices.getUniqueFamilies() };
173
174 float queuePriority{ 1.f };
175 for(auto queueFamily : uniqueQueueFamilies)
176 {
177 VkDeviceQueueCreateInfo queueCreateInfo{
178 .sType = VK_STRUCTURE_TYPE_DEVICE_QUEUE_CREATE_INFO,
179 .queueFamilyIndex = queueFamily,
180 .queueCount = 1,
181 .pQueuePriorities = &queuePriority
182 };
183
184 queueCreateInfos.push_back(queueCreateInfo);
185 }
186
187 VkPhysicalDeviceFeatures deviceFeatures{
188 .samplerAnisotropy = VK_TRUE
189 };
190
191 VkDeviceCreateInfo createInfo{
192 .sType = VK_STRUCTURE_TYPE_DEVICE_CREATE_INFO,
193 .queueCreateInfoCount = static_cast<std::uint32_t>(queueCreateInfos.size()),
194 .pQueueCreateInfos = queueCreateInfos.data(),
195 .enabledLayerCount = 0,
196 .ppEnabledLayerNames = nullptr,
197 .enabledExtensionCount = static_cast<std::uint32_t>(deviceExtensions.size()),
198 .ppEnabledExtensionNames = deviceExtensions.data(),
199 .pEnabledFeatures = &deviceFeatures
200 };
201
202 if(useValidationLayers)
203 {
204 createInfo.enabledLayerCount = static_cast<std::uint32_t>(validationLayers.size());
205 createInfo.ppEnabledLayerNames = validationLayers.data();
206 }
207
208 if(vkCreateDevice(m_physicalDevice, &createInfo, nullptr, &m_device) != VK_SUCCESS)
209 throwWithLog<VulkanException>(std::source_location::current(), VulkanExceptionCause::CREATE_DEVICE);
210
211 if(indices.graphicsFamily.has_value() && indices.presentFamily.has_value())
212 {
213 vkGetDeviceQueue(m_device, indices.graphicsFamily.value(), 0, &m_graphicsQueue);
214 vkGetDeviceQueue(m_device, indices.presentFamily.value(), 0, &m_presentQueue);
215 }
216 else
217 throwWithLog<VulkanException>(std::source_location::current(), VulkanExceptionCause::QUEUE_FAMILY_INDEX_IS_EMPTY);
218
219 spdlog::info("Logical device created successfully...");
220}

230bool VulkanDevice::isPhysicalDeviceSuitable(VkPhysicalDevice device) const
231{
232 const QueueFamilyIndices indices{ findQueueFamilies(device) };
233 const bool extensionsSupported{ checkDeviceExtensionsSupported(device) };

240 }

254QueueFamilyIndices VulkanDevice::findQueueFamilies(VkPhysicalDevice device) const
255{
256 QueueFamilyIndices indices;
257
258 std::uint32_t queueFamilyCount{0};
259 vkGetPhysicalDeviceQueueFamilyProperties(device, &queueFamilyCount, nullptr);
260
261 std::vector<VkQueueFamilyProperties> queueFamilies(queueFamilyCount);
262 vkGetPhysicalDeviceQueueFamilyProperties(device, &queueFamilyCount, queueFamilies.data());
263
264 int i{0};
265 for(const auto& queueFamily : queueFamilies)
266 {
267 if(queueFamily.queueCount > 0 && static_cast<bool>(queueFamily.queueFlags & VK_QUEUE_GRAPHICS_BIT))
268 indices.graphicsFamily.emplace(i);
269
270 VkBool32 presentSupport{ VK_FALSE };
271 vkGetPhysicalDeviceSurfaceSupportKHR(device, i, surface, &presentSupport);
272 if(queueFamily.queueCount > 0 && static_cast<bool>(presentSupport))
273 indices.presentFamily.emplace(i);
274
275 if(indices.isComplete())
276 break;
277
278 ++i;
279 }
280
281 return indices;
282}

290{
291 std::uint32_t extensionCount{0};
292 vkEnumerateDeviceExtensionProperties(device, nullptr, &extensionCount, nullptr);
293

295 vkEnumerateDeviceExtensionProperties(device, nullptr, &extensionCount, availableExtensions.data());
296
297 std::set<std::string, std::less<>> requiredExtensions(deviceExtensions.begin(), deviceExtensions.end());
